// ========== САПЁР ========== let sapperBet = 10; let sapperBombs = 9; let sapperGrid = []; let sapperRevealed = []; let sapperGameOver = false; let sapperFirstClick = true; let sapperCurrentWin = 0; // текущий накопленный выигрыш const sapperMultipliers = { 9: 1.6, 12: 2.2, 15: 3.0, 20: 5.0 }; function openSapper() { document.getElementById('sapperModal').classList.add('active'); resetSapper(); } function closeSapper() { document.getElementById('sapperModal').classList.remove('active'); } function resetSapper() { sapperGrid = []; sapperRevealed = []; sapperGameOver = false; sapperFirstClick = true; sapperCurrentWin = 0; document.getElementById('sapperStatus').textContent = '💰 Выигрыш: 0 Gold'; document.getElementById('sapperStatus').style.color = '#ffd700'; document.getElementById('sapperTakeBtn').disabled = true; // Ставка из поля ввода const betInput = document.getElementById('sapperBetInput'); sapperBet = parseInt(betInput.value) || 10; if (sapperBet < 1) sapperBet = 1; betInput.value = sapperBet; // Сложность document.querySelectorAll('#sapperDifficulty button').forEach(b => b.classList.remove('active')); document.querySelector('#sapperDifficulty button[data-bombs="9"]').classList.add('active'); sapperBombs = 9; renderSapperGrid(); } function renderSapperGrid() { const grid = document.getElementById('sapperGrid'); grid.innerHTML = ''; for (let i = 0; i < 25; i++) { const cell = document.createElement('div'); cell.className = 'sapper-cell'; cell.dataset.index = i; cell.textContent = '?'; cell.addEventListener('click', () => sapperClick(i)); grid.appendChild(cell); } sapperGrid = new Array(25).fill(0); sapperRevealed = new Array(25).fill(false); } function sapperClick(index) { if (sapperGameOver || sapperRevealed[index]) return; if (sapperFirstClick) { sapperFirstClick = false; // Размещаем бомбы (не на первой клетке) let bombsPlaced = 0; while (bombsPlaced < sapperBombs) { const pos = Math.floor(Math.random() * 25); if (pos !== index && sapperGrid[pos] !== 1) { sapperGrid[pos] = 1; bombsPlaced++; } } // Проверяем, не попали ли мы на бомбу при первом клике (безопасно) if (sapperGrid[index] === 1) { sapperGrid[index] = 0; let newPos = Math.floor(Math.random() * 25); while (newPos === index || sapperGrid[newPos] === 1) { newPos = Math.floor(Math.random() * 25); } sapperGrid[newPos] = 1; } // Списываем ставку if (balance < sapperBet) { alert('❌ Недостаточно Gold!'); sapperGameOver = true; document.getElementById('sapperStatus').textContent = '❌ Недостаточно Gold!'; document.getElementById('sapperStatus').style.color = '#ff2d55'; return; } balance -= sapperBet; document.getElementById('balanceDisplay').textContent = balance; document.getElementById('sapperTakeBtn').disabled = false; } if (sapperGrid[index] === 1) { sapperGameOver = true; revealAll(); document.getElementById('sapperStatus').textContent = `💥 БОМБА! Проиграно ${sapperBet} Gold`; document.getElementById('sapperStatus').style.color = '#ff2d55'; document.getElementById('sapperTakeBtn').disabled = true; return; } sapperRevealed[index] = true; const cell = document.querySelector(`.sapper-cell[data-index="${index}"]`); cell.classList.add('revealed', 'safe'); cell.textContent = '✅'; cell.style.color = '#4b69ff'; // Начисляем Gold за клетку (ставка / 10) const perCell = Math.floor(sapperBet / 10); if (perCell > 0) { sapperCurrentWin += perCell; } else { sapperCurrentWin += 1; // минимум 1 Gold за клетку } // Обновляем статус const safeCells = sapperGrid.filter(c => c === 0).length; const revealedSafe = sapperRevealed.filter((r, i) => r && sapperGrid[i] === 0).length; document.getElementById('sapperStatus').textContent = `💰 Выигрыш: ${sapperCurrentWin} Gold (${revealedSafe}/${safeCells} клеток)`; document.getElementById('sapperStatus').style.color = '#ffd700'; // Проверка победы (все клетки открыты) if (revealedSafe === safeCells) { sapperGameOver = true; const winAmount = Math.floor(sapperBet * sapperMultipliers[sapperBombs]); balance += winAmount; document.getElementById('balanceDisplay').textContent = balance; document.getElementById('sapperStatus').textContent = `🎉 ПОБЕДА! Вы выиграли ${winAmount} Gold (x${sapperMultipliers[sapperBombs]})!`; document.getElementById('sapperStatus').style.color = '#ffd700'; revealAll(); document.getElementById('sapperTakeBtn').disabled = true; return; } } function takeWin() { if (sapperGameOver || sapperCurrentWin === 0) return; balance += sapperCurrentWin; document.getElementById('balanceDisplay').textContent = balance; document.getElementById('sapperStatus').textContent = `✅ Вы забрали ${sapperCurrentWin} Gold!`; document.getElementById('sapperStatus').style.color = '#4b69ff'; sapperGameOver = true; revealAll(); document.getElementById('sapperTakeBtn').disabled = true; } function revealAll() { document.querySelectorAll('.sapper-cell').forEach(cell => { const index = parseInt(cell.dataset.index); if (!sapperRevealed[index]) { cell.classList.add('revealed'); if (sapperGrid[index] === 1) { cell.classList.add('bomb'); cell.textContent = '💣'; } else { cell.classList.add('safe'); cell.textContent = '✅'; } } }); } // Выбор ставки document.getElementById('sapperBetInput').addEventListener('change', function() { if (!sapperFirstClick) return; let val = parseInt(this.value) || 10; if (val < 1) val = 1; this.value = val; sapperBet = val; resetSapper(); document.querySelector('#sapperBetInput').value = sapperBet; }); // Выбор сложности document.querySelectorAll('#sapperDifficulty button').forEach(btn => { btn.addEventListener('click', function() { if (!sapperFirstClick) return; document.querySelectorAll('#sapperDifficulty button').forEach(b => b.classList.remove('active')); this.classList.add('active'); sapperBombs = parseInt(this.dataset.bombs); resetSapper(); document.querySelector(`#sapperDifficulty button[data-bombs="${sapperBombs}"]`).classList.add('active'); }); });
